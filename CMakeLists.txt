cmake_minimum_required(VERSION 3.20)
project(Pi5AoIP VERSION 2.3.1 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Pi5-AoIP requires Linux; the runtime additionally checks Pi 5 and PREEMPT_RT")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(cmake/AoIPDependency.cmake)
add_library(aoip_rpi5 STATIC src/rpi5.cpp)
add_library(Pi5AoIP::platform ALIAS aoip_rpi5)
add_library(PiAoIP::rpi5 ALIAS aoip_rpi5)
set_target_properties(aoip_rpi5 PROPERTIES EXPORT_NAME platform POSITION_INDEPENDENT_CODE ON)
target_include_directories(aoip_rpi5 PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(aoip_rpi5 PUBLIC AoIP::peer)
add_executable(aoip_peer_rpi5 apps/main.cpp)
target_link_libraries(aoip_peer_rpi5 PRIVATE Pi5AoIP::platform)
set_target_properties(aoip_peer_rpi5 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS aoip_rpi5 EXPORT Pi5AoIPTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT Pi5AoIPTargets NAMESPACE Pi5AoIP:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Pi5AoIP)
configure_package_config_file(cmake/Pi5AoIPConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/Pi5AoIPConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Pi5AoIP)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/Pi5AoIPConfigVersion.cmake" COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/Pi5AoIPConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/Pi5AoIPConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Pi5AoIP)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/Pi5-AoIP)
